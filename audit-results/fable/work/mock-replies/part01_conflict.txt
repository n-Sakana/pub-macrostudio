Option Explicit

' Win32 API を使わずに待つための共通モジュールです。
Public Sub WaitMilliseconds(ByVal milliseconds As Long)
    Dim startAt As Double
    Dim finishAt As Double

    If milliseconds <= 0 Then Exit Sub
    startAt = Timer
    finishAt = startAt + (milliseconds / 1000#)
    Do While Timer < finishAt
        DoEvents
        If Timer < startAt Then Exit Do
    Loop
End Sub
