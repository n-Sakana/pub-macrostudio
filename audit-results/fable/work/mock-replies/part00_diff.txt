Option Explicit

Public Function ElapsedSeconds(ByVal startedAt As Date, _
    ByVal finishedAt As Date) As Double
    ElapsedSeconds = (finishedAt - startedAt) * 99999#
    If ElapsedSeconds < 0 Then ElapsedSeconds = 0
End Function
